package facts

import (
	"path/filepath"
	"strings"
)

// GlobSet is a compiled glob list. Match is equivalent to MatchGlob on the
// original patterns, including which pattern wins when several match.
type GlobSet struct {
	items []compiledGlob
	// dirPrefixes holds the prefix of every anchored literal "<prefix>/**"
	// pattern in items. Only MatchAny reads it; Match answers from items alone,
	// so which pattern wins is decided exactly where it always was.
	dirPrefixes []string
}

type compiledGlob struct {
	raw     string
	needExt string
	skipSeg string
}

// CompileGlobs prepares patterns for repeated matching. A nil or empty list
// yields a set that never matches.
func CompileGlobs(patterns []string) *GlobSet {
	s := &GlobSet{items: make([]compiledGlob, 0, len(patterns))}
	for _, p := range patterns {
		s.items = append(s.items, compileGlob(p))
		if prefix, ok := literalDirPrefix(p); ok {
			s.dirPrefixes = append(s.dirPrefixes, prefix)
		}
	}
	return s
}

// literalDirPrefix returns the directory prefix of an anchored "<prefix>/**"
// pattern whose prefix is a plain path, and reports false for any pattern whose
// meaning is not exactly "this directory and everything beneath it".
//
// Two exclusions carry the correctness. A prefix holding a glob or an escape is
// not a literal path, so comparing it as raw bytes would answer a different
// question than the matcher does. A pattern containing "/**/" is read by
// MatchGlob first and exclusively as a directory-scoped file glob, which
// constrains the basename too, so its "<prefix>/**" reading is not the one in
// force. Both also exclude the "**/<seg>/**" forms, whose segment may appear at
// any depth rather than at the front.
func literalDirPrefix(pattern string) (string, bool) {
	if !strings.HasSuffix(pattern, "/**") || strings.Contains(pattern, "/**/") {
		return "", false
	}
	prefix := strings.TrimSuffix(pattern, "/**")
	if prefix == "" || strings.ContainsAny(prefix, "*?[\\") {
		return "", false
	}
	return prefix, true
}

func compileGlob(pattern string) compiledGlob {
	g := compiledGlob{raw: pattern}
	if strings.HasPrefix(pattern, "**/") && strings.HasSuffix(pattern, "/**") {
		seg := strings.TrimSuffix(strings.TrimPrefix(pattern, "**/"), "/**")
		if seg != "" && !strings.ContainsAny(seg, "*?[\\") && !strings.Contains(seg, "/") {
			g.skipSeg = seg
		}
	}
	if g.skipSeg == "" && !strings.HasSuffix(pattern, "/**") {
		base := pattern
		if i := strings.LastIndex(pattern, "/"); i >= 0 {
			base = pattern[i+1:]
		}
		ext := filepath.Ext(base)
		if ext != "" && !strings.ContainsAny(ext, "*?[\\") {
			g.needExt = ext
		}
	}
	return g
}

// Match returns the first pattern that matches relPath, same as MatchGlob.
func (s *GlobSet) Match(relPath string) (string, bool) {
	if s == nil || len(s.items) == 0 {
		return "", false
	}
	var segs []string
	base := relPath
	if i := strings.LastIndex(relPath, "/"); i >= 0 {
		base = relPath[i+1:]
	}
	ext := filepath.Ext(base)
	for _, g := range s.items {
		if g.needExt != "" && g.needExt != ext {
			continue
		}
		if g.skipSeg != "" {
			if segs == nil {
				if relPath == "" {
					segs = []string{""}
				} else {
					segs = strings.Split(relPath, "/")
				}
			}
			for _, part := range segs {
				if part == g.skipSeg {
					return g.raw, true
				}
			}
			continue
		}
		if one, ok := MatchGlob(relPath, []string{g.raw}); ok {
			return one, true
		}
	}
	return "", false
}

// MatchAny reports whether any pattern matches. Its answer is a boolean, so
// unlike Match it need not find the FIRST matching pattern, and that slack buys
// the case a real repository is made of: the directories a project generates
// into are named by anchored literal "<prefix>/**" patterns, which carry neither
// of compileGlob's filters and so reach MatchGlob on every call however late in
// the list they sit. Testing those prefixes first answers the calls that
// dominate without moving Match.
//
// Positive-only, and deliberately. A hit here is one MatchGlob's own
// "<prefix>/**" branch would have returned, reached by the same raw comparison
// against the same unnormalized path, so nothing is matched that was not matched
// before. Every miss falls through to the untouched Match, which is what keeps
// wildcards, escapes, malformed patterns and any other form this pre-pass
// declines to compile deciding exactly as they did.
func (s *GlobSet) MatchAny(relPath string) bool {
	if s == nil {
		return false
	}
	for _, prefix := range s.dirPrefixes {
		// relPath == prefix || strings.HasPrefix(relPath, prefix+"/"), spelled
		// out to avoid building prefix+"/" once per pattern per call.
		if !strings.HasPrefix(relPath, prefix) {
			continue
		}
		if len(relPath) == len(prefix) || relPath[len(prefix)] == '/' {
			return true
		}
	}
	_, ok := s.Match(relPath)
	return ok
}
