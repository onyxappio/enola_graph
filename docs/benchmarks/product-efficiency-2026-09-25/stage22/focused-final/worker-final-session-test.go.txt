// These are the end-to-end statements about a minified framework producer: a
// generated gRPC client stub that is never parsed, but whose services a consumer
// resolves against. What the extractor now records on that skipped file is only
// worth anything if a whole session reuses it and still lands on the cold graph,
// refuses it when the bytes move under the run, and upgrades a state written
// before the field existed without hiding what it had to redo.

var sessionMinifiedPad = "\nexport const pad=[" + minifiedPadBody + "0];\n"

// minifiedPadBody is a single very long line, which is what the extractor's
// minified heuristic keys on; it declares nothing the framework collectors read.
var minifiedPadBody = strings.Repeat("0,", 2000)

const sessionUsersService = `// @generated by protobuf-ts
import { ServiceType } from "@protobuf-ts/runtime-rpc";
export const UserService = new ServiceType("users.v1.UserService", [
    { name: "GetUser", options: {}, I: GetUserRequest, O: GetUserResponse },
    { name: "CreateUser", options: {}, I: CreateUserRequest, O: CreateUserResponse }
]);
`

const sessionUsersClient = `// @generated by protobuf-ts
import { UserService } from "./users";
import { stackIntercept } from "@protobuf-ts/runtime-rpc";
/**
 * @generated from protobuf service users.v1.UserService
 */
export class UserServiceClient implements IUserServiceClient {
    typeName = UserService.typeName;
    methods = UserService.methods;
    constructor(private readonly _transport) {}
    /**
     * @generated from protobuf rpc: GetUser(users.v1.GetUserRequest) returns (users.v1.GetUserResponse);
     */
    getUser(input, options) {
        const method = this.methods[0];
        return stackIntercept("unary", this._transport, method, options, input);
    }
    /**
     * @generated from protobuf rpc: CreateUser(users.v1.CreateUserRequest) returns (users.v1.CreateUserResponse);
     */
    createUser(input, options) {
        const method = this.methods[1];
        return stackIntercept("unary", this._transport, method, options, input);
    }
}
`

func minifiedStubRepo(t *testing.T, extra map[string]string) (string, string) {
	t.Helper()
	stub := "gen/users.client.ts"
	files := map[string]string{
		"gen/users.ts": sessionUsersService,
		stub:           sessionUsersClient + sessionMinifiedPad,
		"app.ts": "import { UserServiceClient } from './gen/users.client';\n" +
			"const userService = new UserServiceClient(makeTransport());\n" +
			"export async function submit(name: string) { return await userService.createUser({ name }); }\n",
	}
	for rel, body := range extra {
		files[rel] = body
	}
	return setupTSRepo(t, files), stub
}

func readState(t *testing.T, stateDir string) *State {
	t.Helper()
	raw, err := os.ReadFile(statePath(stateDir))
	if err != nil {
		t.Fatal(err)
	}
	var st State
	if err := json.Unmarshal(raw, &st); err != nil {
		t.Fatal(err)
	}
	return &st
}

func writeState(t *testing.T, stateDir string, st *State) {
	t.Helper()
	raw, err := json.Marshal(st)
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(statePath(stateDir), raw, 0o600); err != nil {
		t.Fatal(err)
	}
}

// assertStubRecorded is the guard that keeps every test below non-vacuous: if
// the fixture stopped being classified minified, or stopped carrying a gRPC
// summary, these tests would be about an ordinary parsed file.
func assertStubRecorded(t *testing.T, st *State, stub string) {
	t.Helper()
	fs := st.Files[stub]
	if fs == nil || !fs.Minified {
		t.Fatalf("%s is not recorded as minified: %+v", stub, fs)
	}
	if fs.TS == nil || fs.TS.GRPC == nil || len(fs.TS.GRPC.Methods) == 0 {
		t.Fatalf("%s carries no gRPC summary: %+v", stub, fs.TS)
	}
	if fs.TS.NuxtScope == "" {
		t.Fatalf("%s carries no scope, so it is still a legacy record: %+v", stub, fs.TS)
	}
}

// A delta hop that reuses the minified record, then a no-op, must land on the
// graph a cold run of the same tree produces, and the no-op must stay silent.
func TestMinifiedFrameworkProducerDeltaAndNoopMatchCold(t *testing.T) {
	dir, stub := minifiedStubRepo(t, nil)
	eng := testEngine(t, dir)
	state := filepath.Join(dir, ".enola", "state")
	cons := NewConsumer()

	s := &graphstream.MemorySink{}
	if _, err := Run(context.Background(), eng, dir, s, Options{AuthoritativeFiles: true, StateDir: state}); err != nil {
		t.Fatal(err)
	}
	applyRun(t, cons, s)
	assertStubRecorded(t, readState(t, state), stub)

	// A body edit on the consumer: the stub is untouched and must be reused.
	if err := os.WriteFile(filepath.Join(dir, "app.ts"), []byte(
		"import { UserServiceClient } from './gen/users.client';\n"+
			"const userService = new UserServiceClient(makeTransport());\n"+
			"export async function submit(name: string) { return await userService.createUser({ name }); }\n"+
			"export async function fetchOne(id: string) { return await userService.getUser({ id }); }\n"), 0o644); err != nil {
		t.Fatal(err)
	}
	s = &graphstream.MemorySink{}
	delta, err := Run(context.Background(), eng, dir, s, Options{AuthoritativeFiles: true, StateDir: state})
	if err != nil {
		t.Fatal(err)
	}
	applyRun(t, cons, s)
	t.Logf("delta parsed=%d fallbacks=%v", delta.ParsedFiles, delta.Fallbacks)
	assertStubRecorded(t, readState(t, state), stub)

	// The no-op after it stays silent.
	s = &graphstream.MemorySink{}
	noop, err := Run(context.Background(), eng, dir, s, Options{AuthoritativeFiles: true, StateDir: state})
	if err != nil {
		t.Fatal(err)
	}
	if noop.ParsedFiles != 0 || noop.TargetGeneration != delta.TargetGeneration || len(s.CloneRecords()) != 0 {
		t.Fatalf("no-op after a minified delta was not silent: parsed=%d gen=%d->%d events=%d",
			noop.ParsedFiles, delta.TargetGeneration, noop.TargetGeneration, len(s.CloneRecords()))
	}

	cold := NewConsumer()
	s = &graphstream.MemorySink{}
	if _, err := Run(context.Background(), eng, dir, s, Options{AuthoritativeFiles: true, StateDir: filepath.Join(dir, ".enola", "cold")}); err != nil {
		t.Fatal(err)
	}
	applyRun(t, cold, s)
	assertAppliedEqualsCold(t, cons, cold)
}

// A Nuxt layout change is a real config change, not an edited cached field: a
// new nested Nuxt package takes ownership of the directory the stub lives in, so
// the stub's durable scope moves and the record must be re-read under it.
func TestMinifiedFrameworkProducerFollowsNuxtLayoutChange(t *testing.T) {
	dir, stub := minifiedStubRepo(t, map[string]string{
		"nuxt.config.ts": "export default defineNuxtConfig({})\n",
		"package.json":   `{"name":"app","type":"module","dependencies":{"nuxt":"^3.0.0"}}`,
	})
	eng := testEngine(t, dir)
	state := filepath.Join(dir, ".enola", "state")
	cons := NewConsumer()

	s := &graphstream.MemorySink{}
	if _, err := Run(context.Background(), eng, dir, s, Options{AuthoritativeFiles: true, StateDir: state}); err != nil {
		t.Fatal(err)
	}
	applyRun(t, cons, s)
	before := readState(t, state)
	assertStubRecorded(t, before, stub)
	was := before.Files[stub].TS.NuxtScope

	// gen/ becomes its own Nuxt application.
	for rel, body := range map[string]string{
		"gen/nuxt.config.ts": "export default defineNuxtConfig({})\n",
		"gen/package.json":   `{"name":"gen","type":"module","dependencies":{"nuxt":"^3.0.0"}}`,
	} {
		if err := os.WriteFile(filepath.Join(dir, rel), []byte(body), 0o644); err != nil {
			t.Fatal(err)
		}
	}
	s = &graphstream.MemorySink{}
	moved, err := Run(context.Background(), eng, dir, s, Options{AuthoritativeFiles: true, StateDir: state})
	if err != nil {
		t.Fatal(err)
	}
	applyRun(t, cons, s)
	t.Logf("layout change parsed=%d fallbacks=%v", moved.ParsedFiles, moved.Fallbacks)

	after := readState(t, state)
	assertStubRecorded(t, after, stub)
	if now := after.Files[stub].TS.NuxtScope; now == was {
		t.Fatalf("the layout moved but the stub's scope did not: still %q", now)
	}

	cold := NewConsumer()
	s = &graphstream.MemorySink{}
	if _, err := Run(context.Background(), eng, dir, s, Options{AuthoritativeFiles: true, StateDir: filepath.Join(dir, ".enola", "cold")}); err != nil {
		t.Fatal(err)
	}
	applyRun(t, cold, s)
	assertAppliedEqualsCold(t, cons, cold)
}

// The reuse is only safe because the run still re-proves the bundle's bytes
// before it commits. A minified file is never parsed, so it reaches no
// OnBeforeParse of its own; the consumer's parse is the moment inside the frozen
// plan at which its bytes are rewritten underneath the run.
func TestMinifiedBytesRewrittenAfterFreezeFailTheRun(t *testing.T) {
	dir, stub := minifiedStubRepo(t, nil)
	eng := testEngine(t, dir)
	state := filepath.Join(dir, ".enola", "state")

	s := &graphstream.MemorySink{}
	if _, err := Run(context.Background(), eng, dir, s, Options{AuthoritativeFiles: true, StateDir: state}); err != nil {
		t.Fatal(err)
	}
	assertStubRecorded(t, readState(t, state), stub)
	firstGeneration := readState(t, state).Generation

	if err := os.WriteFile(filepath.Join(dir, "app.ts"), []byte(
		"import { UserServiceClient } from './gen/users.client';\n"+
			"const userService = new UserServiceClient(makeTransport());\n"+
			"export async function fetchOne(id: string) { return await userService.getUser({ id }); }\n"), 0o644); err != nil {
		t.Fatal(err)
	}
	var rewrote bool
	mutated := &graphstream.MemorySink{}
	opts := Options{AuthoritativeFiles: true, StateDir: state, OnBeforeParse: func(string) {
		if rewrote {
			return
		}
		rewrote = true
		// The rewrite only means something if the plan is already frozen, so
		// prove the full authoritative Begin was emitted before this parse and
		// that nothing has been ended yet.
		begins, _, ends, err := DecodeRun(mutated.CloneRecords())
		if err != nil || len(begins) != 1 || len(ends) != 0 {
			t.Errorf("bytes rewritten outside a frozen plan: begins=%d ends=%d err=%v", len(begins), len(ends), err)
		}
		if err := os.WriteFile(filepath.Join(dir, stub), []byte(
			strings.Replace(sessionUsersClient, "users.v1.UserService", "other.v2.UserService", 1)+sessionMinifiedPad), 0o644); err != nil {
			t.Fatal(err)
		}
	}}
	_, err := Run(context.Background(), eng, dir, mutated, opts)
	if !rewrote {
		t.Fatal("the hook never fired, so nothing was rewritten under the run")
	}
	if !errors.Is(err, ErrInputsChanged) {
		t.Fatalf("a minified bundle rewritten inside the frozen plan was not refused as changed inputs: %v", err)
	}
	// Refused means refused on the wire too: a Begin may stand, but no EndReplace
	// may have succeeded, and the committed generation may not move.
	begins, _, ends, derr := DecodeRun(mutated.CloneRecords())
	if derr != nil {
		t.Fatal(derr)
	}
	if len(begins) != 1 || len(ends) != 0 {
		t.Fatalf("refused run still ended a replacement: begins=%d ends=%d", len(begins), len(ends))
	}
	if got := readState(t, state).Generation; got != firstGeneration {
		t.Fatalf("state advanced despite the refusal: %d -> %d", firstGeneration, got)
	}
}

// A state written before minified records carried a summary has an empty scope
// and no gRPC record. Two things are true about it and both are asserted here.
//
// A run with nothing dirty does not upgrade it: the session persists the
// previous records wholesale on that path (session.go, tsRecords = prevRecs), so
// the refreshed record the extractor derives is not what gets stored. Until the
// file next falls in a changed scope, the signature collector keeps re-reading
// the bundle every run - which is exactly what main already does for it, so this
// is the unchanged old cost rather than a new one.
//
// The first run that does touch its scope upgrades it, lands on the cold graph,
// and must not hide whatever it broadened to get there.
func TestLegacyMinifiedStateUpgradesOnFirstChangedRun(t *testing.T) {
	dir, stub := minifiedStubRepo(t, nil)
	eng := testEngine(t, dir)
	state := filepath.Join(dir, ".enola", "state")
	cons := NewConsumer()

	s := &graphstream.MemorySink{}
	if _, err := Run(context.Background(), eng, dir, s, Options{AuthoritativeFiles: true, StateDir: state}); err != nil {
		t.Fatal(err)
	}
	applyRun(t, cons, s)
	current := readState(t, state)
	assertStubRecorded(t, current, stub)

	// Roll the stub's record back to the shape the previous version wrote: the
	// file's bytes and hash are untouched, only what the extractor recorded
	// about them is removed.
	legacy := readState(t, state)
	rec := *legacy.Files[stub].TS
	rec.GRPC = nil
	rec.GraphQLServer = false
	rec.GraphQLSDL = nil
	rec.NuxtScope = ""
	legacy.Files[stub].TS = &rec
	// The signature the previous version stored alongside such a record is not
	// the candidate's. It is the one the composition collector produces when the
	// minified file contributes nothing, which is exactly how the previous
	// version consumed an incomplete record. Derived this way it reproduces the
	// framework_sig a real Stage20 baseline binary wrote into old-delta-state.json
	// bit for bit, so this fixture carries an old-writer signature rather than
	// the candidate's carried forward.
	legacySig := legacyFrameworkSig(t, dir, legacy, stub)
	if legacySig == current.FrameworkSig {
		t.Fatal("the derived legacy signature equals the candidate's, so this fixture would prove no migration")
	}
	legacy.FrameworkSig = legacySig
	writeState(t, state, legacy)

	// A quiet run leaves it legacy, and stays quiet doing so.
	s = &graphstream.MemorySink{}
	quiet, err := Run(context.Background(), eng, dir, s, Options{AuthoritativeFiles: true, StateDir: state})
	if err != nil {
		t.Fatal(err)
	}
	if quiet.ParsedFiles != 0 || len(s.CloneRecords()) != 0 {
		t.Fatalf("the legacy state made a quiet run noisy: parsed=%d events=%d", quiet.ParsedFiles, len(s.CloneRecords()))
	}
	if still := readState(t, state).Files[stub].TS; still.GRPC != nil || still.NuxtScope != "" {
		t.Fatalf("a quiet run upgraded the record; this test documents that it does not: %+v", still)
	}

	// The first run whose scope includes the stub adopts the refreshed record.
	if err := os.WriteFile(filepath.Join(dir, "app.ts"), []byte(
		"import { UserServiceClient } from './gen/users.client';\n"+
			"const userService = new UserServiceClient(makeTransport());\n"+
			"export async function fetchOne(id: string) { return await userService.getUser({ id }); }\n"), 0o644); err != nil {
		t.Fatal(err)
	}
	s = &graphstream.MemorySink{}
	upgrade, err := Run(context.Background(), eng, dir, s, Options{AuthoritativeFiles: true, StateDir: state})
	if err != nil {
		t.Fatal(err)
	}
	applyRun(t, cons, s)
	t.Logf("legacy upgrade parsed=%d fallbacks=%v", upgrade.ParsedFiles, upgrade.Fallbacks)

	upgraded := readState(t, state)
	assertStubRecorded(t, upgraded, stub)
	// The upgrade is a refresh of what was always derivable from these bytes,
	// so it must not move the recorded hash.
	if upgraded.Files[stub].Hash != current.Files[stub].Hash {
		t.Fatalf("the upgrade changed the recorded hash: %s -> %s",
			current.Files[stub].Hash, upgraded.Files[stub].Hash)
	}
	// The run read an old signature and re-derived a different one, so it
	// broadened its own scope and has to say so - with the reason that actually
	// applies, not merely some fallback that could have come from anywhere else.
	if upgraded.FrameworkSig == legacySig {
		t.Fatalf("the framework signature never migrated off the legacy value %s", legacySig)
	}
	const wantReason = "graphql/grpc/nuxt composition context changed"
	var reasons []string
	var reported bool
	for _, f := range upgrade.Fallbacks {
		reasons = append(reasons, f.Reason)
		if strings.Contains(f.Reason, wantReason) {
			reported = true
		}
	}
	if !reported {
		t.Fatalf("the signature moved %s -> %s without reporting %q; fallbacks: %v",
			legacySig, upgraded.FrameworkSig, wantReason, reasons)
	}

	cold := NewConsumer()
	s = &graphstream.MemorySink{}
	if _, err := Run(context.Background(), eng, dir, s, Options{AuthoritativeFiles: true, StateDir: filepath.Join(dir, ".enola", "cold")}); err != nil {
		t.Fatal(err)
	}
	applyRun(t, cold, s)
	assertAppliedEqualsCold(t, cons, cold)
}

// legacyFrameworkSig derives the framework signature a pre-change writer would
// have stored for this repo: every record exactly as persisted, except that the
// incomplete minified record is given a scope so the collector consumes it as-is
// and it contributes nothing - the pre-change behaviour. The scope only selects
// that path; NuxtScope does not itself participate in the signature.
func legacyFrameworkSig(t *testing.T, dir string, st *State, stub string) string {
	t.Helper()
	var names []string
	prev := map[string]*tsextractor.FileRecord{}
	for rel, fs := range st.Files {
		if fs == nil || fs.TS == nil {
			continue
		}
		names = append(names, rel)
		rec := *fs.TS
		if rel == stub {
			rec.NuxtScope = "-"
		}
		prev[rel] = &rec
	}
	sort.Strings(names)
	sig, err := tsextractor.CompositionSignature(dir, names, prev, map[string]bool{}, nil)
	if err != nil {
		t.Fatal(err)
	}
	return sig
}
